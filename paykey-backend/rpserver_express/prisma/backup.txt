enum Role {
  ADMIN
  USER
}

enum KeyStatus {
  ACTIVE
  SUSPENDED
  REVOKED
}

// ==========================================
// 1. USER CORE
// ==========================================

model User {
  id            String    @id @default(uuid()) @db.VarChar(128)
  email         String    @unique
  fullName      String    @map("full_name")
  passwordHash  String?   @map("password_hash")
  role          Role      @default(USER)
  companyName   String?   @map("company_name")
  mobile        String?   
  createdAt     DateTime  @default(now()) @map("created_at")
  status        String    @default("active")
  suspendNote   String?   @map("suspend_note") @db.Text
  suspendReason String?   @map("suspend_reason") @db.Text
  
  // Relations
  keys           UserKey[]
  authLogs       AuthLog[]
  knownDevices   UserKnownDevice[]
  knownLocations UserKnownLocation[]
  transactions   Transaction[]
  
  // Java Managed Relations
  deviceKeys     AuthDeviceKey[]
  totpTokens     AuthTotpToken[]
  pushRequests   AuthPushRequest[] // Ditambahkan untuk relasi Push Approval
  userDevices    UserDevice[]      // Ditambahkan untuk Physical Device Tracking

  @@map("users")
}

model UserKey {
  id                  BigInt    @id @default(autoincrement())
  aaguid              String    @map("aaguid") @db.VarChar(36)
  attestationType     Int?      @map("attestation_type")
  credentialId        String    @unique @map("credential_id") @db.VarChar(256)
  publicKey           String    @map("public_key") @db.Text
  registeredTimestamp DateTime? @default(now()) @map("registered_timestamp")
  signCounter         BigInt?   @map("sign_counter")
  signatureAlgorithm  Int?      @map("signature_algorithm")
  userDisplayName     String?   @map("user_display_name") @db.VarChar(64)
  userId              String    @map("user_id") @db.VarChar(128)
  username            String?   @db.VarChar(64)
  deviceName          String?   @map("device_name")
  deviceTelemetry     Json?     @map("device_telemetry")
  lastActive          DateTime  @default(now()) @updatedAt @map("last_active")
  lastStatusChange    DateTime  @default(now()) @map("last_status_change")
  lastUsedIp          String?   @map("last_used_ip")
  status              KeyStatus @default(ACTIVE)
  transports          String?
  rpEntityId          String?   @map("rp_entity_id") @db.VarChar(255)
  authenticated_timestamp DateTime? @db.DateTime(6)
  cred_protect        Int?
  rk                  Boolean?  @db.Bit(1)
  user_icon           String?   @db.VarChar(128)
  
  user                User      @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@index([userId])
  @@map("user_key")
}

model AuthDeviceKey {
  id            String    @id @default(uuid())
  userId        String    @map("user_id")
  deviceId      String    @map("device_id")
  
  // FIELD KRUSIAL (JAVA COMPATIBILITY)
  publicKey     String    @map("public_key") @db.Text
  publicKeyHash String?   @map("public_key_hash") @db.VarChar(64)
  algorithm     String?   @db.VarChar(20)

  authType      String    @map("auth_type")
  status        String?   
  createdAt     DateTime  @default(now()) @map("created_at")
  lastUsedAt    DateTime? @map("last_used_at")

  user          User      @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@map("auth_device_keys")
}

model AuthTotpToken {
  id            String    @id @default(uuid())
  serialNumber  String    @unique @map("serial_number")
  
  // FIELD KRUSIAL (JAVA COMPATIBILITY)
  encryptedSeed String    @map("encrypted_seed") @db.VarChar(512)
  timeStep      Int       @default(30) @map("time_step")
  digits        Int       @default(6)
  importedAt    DateTime  @default(now()) @map("imported_at")
  
  userId        String?   @map("user_id")
  tokenType     String?   @map("token_type") 
  status        String?   
  assignedAt    DateTime? @map("assigned_at")
  lastUsedAt    DateTime? @map("last_used_at")

  user          User?     @relation(fields: [userId], references: [id])

  @@map("auth_totp_tokens")
}

model Transaction {
  id               String   @id
  timestamp        DateTime @default(now())
  amount           Float
  currency         String   @default("MYR")
  
  merchantName     String?  @default("Unknown Merchant")
  merchantCategory String? 
  merchantCountry  String?  

  authMethod       String?  
  authResult       String?  
  ipAddress        String?
  locationCity     String?
  locationCountry  String?
  
  userId           String
  user             User     @relation(fields: [userId], references: [id])
  
  deviceInfo       Json?    
  riskLevel        String   @default("LOW")
  riskScore        Int?     @default(0)
  riskReason       String?  @db.Text  

  appliedPolicyId  String?
  appliedLimitId   String?

  riskAnalysisLogs RiskAnalysisLog[]

  @@index([userId])
  @@index([timestamp])
  @@map("transactions")
}

model RiskAnalysisLog {
  id               String      @id @default(uuid())

  ruleCode         String   
  scoreContributed Int      
  
  createdAt        DateTime @default(now())

  @@map("transaction_analysis_logs")
}
